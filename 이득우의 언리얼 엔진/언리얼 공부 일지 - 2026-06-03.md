### 예제 코드 또는 상황
## CPP
```cpp
// 필요한 경우 코드 작성
void UMyGameInstance::Init()  
{  
    Super::Init();  
    // 현재 본인의 클래스 정보  
    UClass* ClassRuntime = GetClass();
    // 위랑 똑같은 객체를 가르킴 
    UClass* ClassCompile = UMyGameInstance::StaticClass();  
    // check는 false를 반환  하면 에디터를 꺼버림  
    check(ClassRuntime != ClassCompile); 
    // ensure을 하면 false를 반환해도 에디터는 안꺼짐   
    ensure(ClassRuntime != ClassCompile); 
    // 에러 메세지에 커스텀 메세지를 포함 시킴
    ensureMsgf(ClassRuntime != ClassCompile ,TEXT("일부러 오류를 일으키는 코드"));    
    UE_LOG(LogTemp, Log, TEXT("학교를 담당하는 클래스 이름 : %s"), 
    *ClassRuntime->GetName());  
    SchoolName = TEXT("++대학교");  
      
    UE_LOG(LogTemp, Log, TEXT("학교 이름 : %s"), *SchoolName);  
    UE_LOG(LogTemp, Log, TEXT("학교 이름 기본값 : %s"),
     *GetClass()->GetDefaultObject<UMyGameInstance>()->SchoolName);  
      
    UE_LOG(LogTemp, Log, TEXT("==========================="));  
}

```
